    41→const formatMarketCap = (num: number): string => {
    42→  if (num >= 1e12) return `${(num / 1e12).toFixed(1)}T`

    44→  if (num >= 1e6) return `${(num / 1e6).toFixed(1)}M`
    45→  return formatNumber(num)

   197→      </div>
   198→    )
   199→  }
   200→  return (
   201→    <img
   202→      src={src}
   203→      alt={alt}
   204→      width={size}
   205→      height={size}
   206→      className={className}
   207→      style={{ minWidth: size, minHeight: size }}
   208→      onError={() => setErrored(true)}
   209→    />

   216→function LoginPage() {
   217→  const [authMode, setAuthMode] = useState<'login' | 'register' | 'pin'>('login')
   218→  const [identifier, setIdentifier] = useState('') // email or phone
   219→  const [password, setPassword] = useState('')
   220→  const [name, setName] = useState('')
   221→  const [email, setEmail] = useState('')
   222→  const [phone, setPhone] = useState('')
   223→  const [confirmPassword, setConfirmPassword] = useState('')
   224→  const [pin, setPin] = useState(['', '', '', '', '', ''])
   225→  const [registerPin, setRegisterPin] = useState(['', '', '', '', '', ''])
   226→  const [confirmPin, setConfirmPin] = useState(['', '', '', '', '', ''])
   227→  const [showPassword, setShowPassword] = useState(false)
   228→  const [showConfirmPassword, setShowConfirmPassword] = useState(false)
   229→  const [loading, setLoading] = useState(false)
   230→  const [agreeTerms, setAgreeTerms] = useState(false)
   231→  const [tempToken, setTempToken] = useState<string | null>(null)
   232→  const login = useAuthStore((s) => s.login)
   233→
   234→  const pinRefs = useRef<(HTMLInputElement | null)[]>([])
   235→  const registerPinRefs = useRef<(HTMLInputElement | null)[]>([])
   236→  const confirmPinRefs = useRef<(HTMLInputElement | null)[]>([])
   237→
   238→  const handlePinChange = (index: number, value: string, pinArr: string[], setPinArr: (v: string[]) => void, refs: React.MutableRefObject<(HTMLInputElement | null)[]>) => {
   239→    if (!/^\d*$/.test(value)) return
   240→    const newPin = [...pinArr]
   241→    newPin[index] = value.slice(-1)
   242→    setPinArr(newPin)
   243→    if (value && index < 5) {
   244→      refs.current[index + 1]?.focus()
   245→    }
   246→  }
   247→
   248→  const handlePinKeyDown = (index: number, e: React.KeyboardEvent, pinArr: string[], setPinArr: (v: string[]) => void, refs: React.MutableRefObject<(HTMLInputElement | null)[]>) => {
   249→    if (e.key === 'Backspace' && !pinArr[index] && index > 0) {
   250→      refs.current[index - 1]?.focus()
   251→    }
   252→  }
   253→
   254→  const handlePinPaste = (e: React.ClipboardEvent, setPinArr: (v: string[]) => void, refs: React.MutableRefObject<(HTMLInputElement | null)[]>) => {
   255→    e.preventDefault()
   256→    const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6)
   257→    const newPin = [...Array(6)].map((_, i) => pasted[i] || '')
   258→    setPinArr(newPin)
   259→    const nextIndex = Math.min(pasted.length, 5)
   260→    refs.current[nextIndex]?.focus()
   261→  }
   262→
   263→  // Step 1: Login with identifier + password
   264→  const handleLoginStep1 = async (e: React.FormEvent) => {
   265→    e.preventDefault()
   266→    if (!identifier || !password) {
   267→      toast({ title: 'Error', description: 'Mohon isi email/nomor HP dan kata sandi', variant: 'destructive' })
   268→      return
   269→    }
   270→    setLoading(true)
   271→    try {
   272→      const res = await fetch('/api/auth/login', {
   273→        method: 'POST',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({ identifier, password })
   276→      })
   277→      const data = await res.json()
   278→      if (!res.ok) throw new Error(data.error)
   279→      if (data.step === 'pin_required') {
   280→        setTempToken(data.tempToken)
   281→        setAuthMode('pin')
   282→        setTimeout(() => pinRefs.current[0]?.focus(), 100)
   283→      } else {
   284→        // Admin direct login (no PIN required for admin)
   285→        if (data.user?.role === 'admin') {
   286→          localStorage.setItem('adminId', data.user.id)
   287→          localStorage.setItem('adminToken', data.token)
   288→          window.location.href = '/admin'
   289→          return
   290→        }
   291→        login(data.user, data.token)
   292→        toast({ title: 'Selamat Datang!', description: `Halo, ${data.user.name}` })
   293→      }
   294→    } catch (err: unknown) {
   295→      toast({ title: 'Gagal', description: err instanceof Error ? err.message : 'Terjadi kesalahan', variant: 'destructive' })
   296→    } finally { setLoading(false) }
   297→  }
   298→
   299→  // Step 2: Verify PIN
   300→  const handlePinVerify = async (e: React.FormEvent) => {
   301→    e.preventDefault()
   302→    const pinStr = pin.join('')
   303→    if (pinStr.length !== 6) {
   304→      toast({ title: 'Error', description: 'Masukkan 6 digit PIN', variant: 'destructive' })
   305→      return
   306→    }
   307→    if (!tempToken) {
   308→      toast({ title: 'Error', description: 'Sesi kadaluarsa, silakan login ulang', variant: 'destructive' })
   309→      setAuthMode('login')
   310→      return
   311→    }
   312→    setLoading(true)
   313→    try {
   314→      const res = await fetch('/api/auth/login', {
   315→        method: 'POST',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify({ tempToken, pin: pinStr })
   318→      })
   319→      const data = await res.json()
   320→      if (!res.ok) throw new Error(data.error)
   321→      if (data.user?.role === 'admin') {
   322→        localStorage.setItem('adminId', data.user.id)
   323→        localStorage.setItem('adminToken', data.token)
   324→        window.location.href = '/admin'
   325→        return
   326→      }
   327→      login(data.user, data.token)
   328→      toast({ title: 'Selamat Datang!', description: `Halo, ${data.user.name}` })
   329→    } catch (err: unknown) {
   330→      toast({ title: 'PIN Salah', description: err instanceof Error ? err.message : 'Verifikasi PIN gagal', variant: 'destructive' })
   331→      setPin(['', '', '', '', '', ''])
   332→      setTimeout(() => pinRefs.current[0]?.focus(), 100)
   333→    } finally { setLoading(false) }
   334→  }
   335→
   336→  // Register
   337→  const handleRegister = async (e: React.FormEvent) => {
   338→    e.preventDefault()
   339→    if (!name || !email || !phone || !password || !confirmPassword) {
   340→      toast({ title: 'Error', description: 'Mohon isi semua field', variant: 'destructive' })
   341→      return
   342→    }
   343→    if (password !== confirmPassword) {
   344→      toast({ title: 'Error', description: 'Kata sandi tidak cocok', variant: 'destructive' })
   345→      return
   346→    }
   347→    if (password.length < 6) {
   348→      toast({ title: 'Error', description: 'Kata sandi minimal 6 karakter', variant: 'destructive' })
   349→      return
   350→    }
   351→    const pinStr = registerPin.join('')
   352→    const confirmPinStr = confirmPin.join('')
   353→    if (pinStr.length !== 6) {
   354→      toast({ title: 'Error', description: 'Masukkan 6 digit PIN', variant: 'destructive' })
   355→      return
   356→    }
   357→    if (pinStr !== confirmPinStr) {
   358→      toast({ title: 'Error', description: 'Konfirmasi PIN tidak cocok', variant: 'destructive' })
   359→      return
   360→    }
   361→    if (!agreeTerms) {
   362→      toast({ title: 'Error', description: 'Anda harus menyetujui proses pendaftaran', variant: 'destructive' })
   363→      return
   364→    }
   365→    setLoading(true)
   366→    try {
   367→      const res = await fetch('/api/auth/register', {
   368→        method: 'POST',
   369→        headers: { 'Content-Type': 'application/json' },
   370→        body: JSON.stringify({ name, email, phone, password, pin: pinStr })
   371→      })
   372→      const data = await res.json()
   373→      if (!res.ok) throw new Error(data.error)
   374→      if (data.user?.role === 'admin') {
   375→        localStorage.setItem('adminId', data.user.id)
   376→        localStorage.setItem('adminToken', data.token)
   377→        window.location.href = '/admin'
   378→        return
   379→      }
   380→      login(data.user, data.token)
   381→      toast({ title: 'Registrasi Berhasil!', description: `Selamat datang, ${data.user.name}!` })
   382→    } catch (err: unknown) {
   383→      toast({ title: 'Gagal', description: err instanceof Error ? err.message : 'Terjadi kesalahan', variant: 'destructive' })
   384→    } finally { setLoading(false) }
   385→  }

  1061→  // Generate a single realistic OHLC candle from previous close
  1062→  // Uses a structured approach: first determine the candle's direction & body,
  1063→  // then add realistic wicks based on intra-candle price action simulation
  1064→  const generateCandle = useCallback((prevClose: number, baseVal: number, sim: {momentum: number; trend: number; phase: number; phaseLen: number; vol: number}, idx: number): CandleData => {
  1065→    // ── Phase Management (longer phases for realistic trends) ──
  1066→    // 0 = consolidation/ranging, 1 = trending, 2 = breakout/volatile
  1067→    sim.phaseLen -= 1
  1068→    if (sim.phaseLen <= 0) {
  1069→      const r = Math.random()
  1070→      if (r < 0.30) {
  1071→        // Consolidation — price chops sideways in a range
  1072→        sim.phase = 0
  1073→        sim.phaseLen = Math.floor(8 + Math.random() * 18) // 8-25 candles of ranging
  1074→      } else if (r < 0.82) {
  1075→        // Trending — sustained directional move (most common in real markets)
  1076→        sim.phase = 1
  1077→        sim.phaseLen = Math.floor(12 + Math.random() * 30) // 12-41 candles of trend
  1078→        // 65% continue existing trend, 35% reverse
  1079→        if (Math.random() < 0.35) sim.trend = (sim.trend === 1 ? -1 : 1) as 1 | -1
  1080→      } else {
  1081→        // Breakout — volatile expansion after consolidation
  1082→        sim.phase = 2
  1083→        sim.phaseLen = Math.floor(3 + Math.random() * 6) // 3-8 volatile candles
  1084→        sim.trend = Math.random() > 0.5 ? 1 : -1
  1085→      }
  1086→    }
  1087→
  1088→    // ── Core Price Movement Engine ──
  1089→    const open = prevClose
  1090→
  1091→    // Volatility scales with phase — trending markets have moderate vol,
  1092→    // consolidation has low vol, breakouts have high vol
  1093→    const baseVol = baseVal * 0.0012
  1094→    const phaseVol = sim.phase === 0 ? 0.5 : sim.phase === 1 ? 1.0 : 2.2
  1095→    const volatility = baseVol * phaseVol
  1096→
  1097→    // Trend drift — the directional force
  1098→    let drift: number
  1099→    if (sim.phase === 1) {
  1100→      // Strong trend: consistent directional bias
  1101→      drift = sim.trend * baseVal * 0.0022
  1102→      // Small pullback within trend (~20% of candles pull back against trend)
  1103→      if (Math.random() < 0.18) {
  1104→        drift = -sim.trend * baseVal * 0.0008
  1105→      }
  1106→    } else if (sim.phase === 2) {
  1107→      // Breakout: strong directional move with high vol
  1108→      drift = sim.trend * baseVal * 0.0035
  1109→    } else {
  1110→      // Consolidation: very small random drift, mostly noise
  1111→      drift = (Math.random() - 0.5) * baseVal * 0.0004
  1112→    }
  1113→
