    82→
    83→  // IHSG chart initialization
    84→  useEffect(() => {
    85→    const ihsgIdx = indices.find(idx => idx.code === 'IHSG') || indices[0]
    86→    if (!ihsgIdx || ihsgChartRef.initialized) return
    87→    ihsgChartRef.initialized = true
    88→    const baseVal = ihsgIdx.value
    89→    ihsgChartRef.baseVal = baseVal
    90→    const isUp = ihsgIdx.changePercent >= 0
    91→    const pts: {idx: number; value: number}[] = []
    92→    let val = baseVal * (1 + (isUp ? -0.005 : 0.005))
    93→    let momentum = 0
    94→    for (let i = 0; i < 50; i++) {
    95→      const dir = Math.random() > 0.5 ? 1 : -1
    96→      const minStep = Math.max(1, baseVal * 0.0005)
    97→      const stepSize = minStep * (0.8 + Math.random() * 1.2)
    98→      const bias = (isUp ? 1 : -1) * baseVal * 0.00005
    99→      momentum = momentum * 0.25 + dir * stepSize + bias
   100→      val += momentum
   101→      val += (baseVal - val) * 0.005
   102→      pts.push({ idx: i, value: Math.round(val) })
   103→    }
   104→    pts.push({ idx: 50, value: baseVal })
   105→    setIhsgChartData(pts)
   106→    ihsgChartRef.val = baseVal
   107→  }, [indices, setIhsgChartData])
   108→
   109→  // IHSG live update
   110→  useEffect(() => {
   111→    let ihsgMomentum = 0
   112→    const interval = setInterval(() => {
   113→      const ref = ihsgChartRef
   114→      if (!ref.initialized) return
   115→      const dir = Math.random() > 0.5 ? 1 : -1
   116→      const stepSize = ref.baseVal * (0.0004 + Math.random() * 0.001)
   117→      ihsgMomentum = ihsgMomentum * 0.25 + dir * stepSize
   118→      ref.val += ihsgMomentum
   119→      ref.val += (ref.baseVal - ref.val) * 0.004
   120→      setIhsgChartData(prev => {
   121→        if (prev.length === 0) return prev
   122→        const nextIdx = prev[prev.length - 1].idx + 1
   123→        const next = [...prev, { idx: nextIdx, value: Math.round(ref.val) }]
   124→        return next.length > 60 ? next.slice(-60) : next
   125→      })
   126→    }, 2500)
   127→    return () => clearInterval(interval)
   128→  }, [setIhsgChartData])
   129→
   130→  // Live sparkline update
   131→  useEffect(() => {
   132→    const interval = setInterval(() => {
   133→      const currentStocks = stocksRef.current
   134→      if (currentStocks.length === 0) return

   136→        const sim = sparklineSimRef.get(s.id)
   137→        const cached = sparklineCache.get(s.id)
   138→        if (!sim || !cached) return
   139→        const dir = Math.random() > 0.5 ? 1 : -1
   140→        const minStep = Math.max(1, s.price * 0.001)
   141→        const stepSize = minStep * (0.5 + Math.random() * 1)
   142→        sim.momentum = sim.momentum * 0.25 + dir * stepSize
   143→        sim.val += sim.momentum
   144→        const reversionStrength = 0.008 + Math.abs(s.price - sim.val) / s.price * 0.08
   145→        sim.val += (s.price - sim.val) * Math.min(reversionStrength, 0.04)
   146→        const newPts = cached.slice(1).map((pt, idx) => ({ i: idx, p: pt.p }))
   147→        newPts.push({ i: cached.length - 1, p: Math.round(sim.val) })
   148→        sparklineCache.set(s.id, newPts)
   149→      })
   150→      setStocks(prev => [...prev])
   151→    }, 2500)
   152→    return () => clearInterval(interval)
   153→  }, [setStocks])
   154→
   155→  // Live price simulation for stock detail
   156→  useEffect(() => {
   157→    if (!selectedStock || !showStockDetail) {
   158→      setLiveChartActive(false)
   159→      return
   160→    }
   161→    const basePrice = selectedStock.price
   162→    const spread = basePrice * 0.002
   163→    let buyPrice = basePrice - spread / 2
   164→    let sellPrice = basePrice + spread / 2
   165→    let momentum = 0
   166→    let phase = 0
   167→    const initialBuy: {time: string; price: number}[] = []
   168→    const initialSell: {time: string; price: number}[] = []
   169→    let tempBuy = buyPrice
   170→    let tempSell = sellPrice
   171→    let histMomentum = 0
   172→    for (let i = 40; i >= 1; i--) {
   173→      const dir = Math.random() > 0.5 ? 1 : -1
   174→      const minStep = Math.max(1, basePrice * 0.001)
   175→      const stepSize = minStep * (0.8 + Math.random() * 1.2)
   176→      histMomentum = histMomentum * 0.25 + dir * stepSize
   177→      const mid = (tempBuy + tempSell) / 2 + histMomentum
   178→      tempBuy = mid - spread / 2
   179→      tempSell = mid + spread / 2
   180→      tempBuy += (basePrice - tempBuy) * 0.005
   181→      tempSell += (basePrice - tempSell) * 0.005
   182→      const now = Date.now() - i * 2000
   183→      const timeStr = new Date(now).toLocaleTimeString('id-ID', {hour: '2-digit', minute: '2-digit', second: '2-digit'})
   184→      initialBuy.push({time: timeStr, price: Math.round(tempBuy)})
   185→      initialSell.push({time: timeStr, price: Math.round(tempSell)})
   186→    }
   187→    buyPrice = tempBuy
   188→    sellPrice = tempSell
   189→    setLiveBuyChart(initialBuy)
   190→    setLiveSellChart(initialSell)
   191→    setLiveBuyPrice(Math.round(buyPrice))
   192→    setLiveSellPrice(Math.round(sellPrice))
   193→    liveChartRef.buyPrice = buyPrice
   194→    liveChartRef.sellPrice = sellPrice
   195→    liveChartRef.trend = 0
   196→    liveChartRef.momentum = 0
   197→    liveChartRef.phase = phase
   198→    setLiveChartActive(true)
   199→    const interval = setInterval(() => {
   200→      phase++
   201→      const dir = Math.random() > 0.5 ? 1 : -1
   202→      const minStep = Math.max(1, basePrice * 0.0008)
   203→      const stepSize = minStep * (0.5 + Math.random() * 1)
   204→      momentum = momentum * 0.25 + dir * stepSize
   205→      const mid = (buyPrice + sellPrice) / 2 + momentum
   206→      buyPrice = mid - spread / 2
   207→      sellPrice = mid + spread / 2
   208→      if (sellPrice <= buyPrice) sellPrice = buyPrice + spread
   209→      buyPrice += (basePrice - buyPrice) * 0.003
   210→      sellPrice += (basePrice - sellPrice) * 0.003
   211→      liveChartRef.buyPrice = buyPrice
   212→      liveChartRef.sellPrice = sellPrice
   213→      liveChartRef.momentum = momentum
   214→      liveChartRef.phase = phase
   215→      const now = new Date()
   216→      const timeStr = now.toLocaleTimeString('id-ID', {hour: '2-digit', minute: '2-digit', second: '2-digit'})
   217→      setLiveBuyChart(prev => { const next = [...prev, {time: timeStr, price: Math.round(buyPrice)}]; return next.length > MAX_CHART_POINTS ? next.slice(-MAX_CHART_POINTS) : next })
   218→      setLiveSellChart(prev => { const next = [...prev, {time: timeStr, price: Math.round(sellPrice)}]; return next.length > MAX_CHART_POINTS ? next.slice(-MAX_CHART_POINTS) : next })
   219→      setLiveBuyPrice(Math.round(buyPrice))
   220→      setLiveSellPrice(Math.round(sellPrice))
   221→    }, 2000)
   222→    return () => { clearInterval(interval); setLiveChartActive(false) }
   223→  }, [selectedStock, showStockDetail, setLiveBuyChart, setLiveSellChart, setLiveBuyPrice, setLiveSellPrice, setLiveChartActive])

   225→  // Background price ticker (when NOT on sinyal tab)
   226→  useEffect(() => {
   227→    if (activeTab === 'sinyal' || !selectedSinyalStock) return

   231→    const interval = setInterval(() => {
   232→      if (sharedChartSimRef) {
   233→        const sim = sharedChartSimRef
   234→        sim.momentum += (Math.random() - 0.5) * sim.vol * 0.002
   235→        sim.momentum *= 0.95
   236→        sim.trend = Math.sin(sim.phase / sim.phaseLen * Math.PI * 2) * 0.4

   243→        setSinyalCurrentPrice(prev => {
   244→          const basePrice = selectedSinyalStock.price
   245→          const currentPrice = prev || basePrice
   246→          const vol = basePrice * 0.0005
   247→          const change = (Math.random() - 0.5) * vol * 2
   248→          return currentPrice + change
   249→        })
   250→      }
   251→    }, tickIntervalMs)
   252→    return () => clearInterval(interval)
   253→  }, [activeTab, selectedSinyalStock, sinyalPositions, setSinyalCurrentPrice])

   258→    if (!initialized.current) { initialized.current = true }

   274→      {/* Top Header */}
   275→      <header className="sticky top-0 z-40 glass-header border-b border-[var(--zv-border)]">

   279→              <Menu className="w-4 h-4 text-[var(--zv-muted)]" />
   280→            </button>
   281→            <ZevorikLogo size={36} className="flex-shrink-0" />

   283→              <b className="block text-[13px] md:text-base font-black gradient-text leading-tight">ZEVORIK</b>

   296→              <Bell className="w-4 h-4 text-[var(--zv-muted)]" />

   298→            </button>
   299→          </div>

   306→          <div className="max-w-7xl mx-auto flex gap-3 px-3 md:px-6 py-2">
   307→            {indices.map(idx => (
   308→              <div key={idx.id} className="flex-shrink-0 flex items-center gap-1.5 bg-[var(--zv-surface)] rounded-lg px-2.5 py-1 border border-[var(--zv-border)]">
   309→                <span className="text-[8px] md:text-[10px] font-black text-[var(--zv-muted)]">{idx.code}</span>
   310→                <span className="text-[10px] md:text-[11px] font-black text-[var(--zv-text)] tabular-nums">{formatNumber(idx.value)}</span>

   312→              </div>
   313→            ))}

   338→
   339→          {/* ====== PROFILE TAB ====== */}

   342→        </AnimatePresence>

   368→                </div>
   369→                <div className="space-y-3 mb-4">
   370→                  <div>

   372→                    <input type="number" value={contractAmount} onChange={e => setContractAmount(e.target.value)} placeholder="Min. Rp 100.000" className="w-full h-10 rounded-xl bg-[var(--zv-surface)] border border-[var(--zv-border)] px-3 text-[12px] font-semibold text-[var(--zv-text)] outline-none focus:border-[#3b82f6] transition-colors" />

   382→                  {contractLoading ? 'Memproses...' : 'Beli Kontrak'}
   383→                </button>
